# TecnoFit Base Import Automation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Criar uma automação vinculada ao Google Sheets que valide os três relatórios HTML com extensão `.xls`, consolide alunos e contratos, atualize as abas de forma recuperável, registre auditoria e arquive cada lote semanal.

**Architecture:** O Apps Script ficará vinculado à planilha mestre e será organizado em módulos pequenos: configuração, parsing, normalização, transformação, acesso ao Drive, escrita no Sheets, auditoria e orquestração. A lógica pura será testada localmente com o test runner nativo do Node; integrações com Drive e Sheets serão verificadas em um lote real controlado antes do uso semanal.

**Tech Stack:** Google Apps Script V8, DriveApp, SpreadsheetApp, Node.js 24 `node:test`, `@google/clasp` 3.3 para sincronização local.

```javascript
function decodeHtmlCell_(value) {
  return String(value || '')
    .replace(/<br\s*\/?\s*>/gi, ' ')
    .replace(/<[^>]+>/g, '')
    .replace(/&nbsp;|&#160;/gi, ' ')
    .replace(/&amp;/gi, '&')
    .replace(/&quot;/gi, '"')
    .replace(/&#39;|&apos;/gi, "'")
    .replace(/&lt;/gi, '<')
    .replace(/&gt;/gi, '>')
    .replace(/&#(\d+);/g, function (_, code) { return String.fromCharCode(Number(code)); })
    .replace(/\s+/g, ' ')
    .trim();
}

function normalizeHeaderForParser_(value) {
  return decodeHtmlCell_(value).normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim();
}

function parseHtmlTable(html) {
  var rowMatches = String(html || '').match(/<tr\b[^>]*>[\s\S]*?<\/tr>/gi) || [];
  return rowMatches.map(function (rowHtml) {
    var cells = rowHtml.match(/<t[dh]\b[^>]*>[\s\S]*?<\/t[dh]>/gi) || [];
    return cells.map(decodeHtmlCell_);
  }).filter(function (row) {
    return row.some(function (cell) { return cell !== ''; });
  });
}

function tableToObjects(rows, requiredHeaders) {
  var wanted = requiredHeaders.map(normalizeHeaderForParser_);
  var headerIndex = rows.findIndex(function (row) {
    var normalized = row.map(normalizeHeaderForParser_);
    return wanted.every(function (header) { return normalized.indexOf(header) !== -1; });
  });
  if (headerIndex === -1) throw new Error('Cabeçalhos obrigatórios não encontrados: ' + wanted.join(', '));
  var normalizedHeaders = rows[headerIndex].map(normalizeHeaderForParser_);
  var positions = {};
  wanted.forEach(function (header) { positions[header] = normalizedHeaders.indexOf(header); });
  return rows.slice(headerIndex + 1).filter(function (row) {
    return wanted.some(function (header) { return decodeHtmlCell_(row[positions[header]]) !== ''; });
  }).map(function (row) {
    var result = {};
    wanted.forEach(function (header) { result[header] = decodeHtmlCell_(row[positions[header]]); });
    return result;
  });
}
```

```javascript
function normalizeId(value) {
  var id = String(value == null ? '' : value).trim().replace(/\.0$/, '');
  if (id && !/^\d+$/.test(id)) throw new Error('ID inválido: ' + value);
  return id;
}

function parseBrDate(value) {
  if (value instanceof Date && !isNaN(value.getTime())) return new Date(value.getTime());
  var text = String(value == null ? '' : value).trim();
  if (!text) return '';
  var match = /^(\d{2})\/(\d{2})\/(\d{4})$/.exec(text);
  if (!match) throw new Error('Data inválida: ' + text);
  var day = Number(match[1]);
  var month = Number(match[2]);
  var year = Number(match[3]);
  var date = new Date(year, month - 1, day, 12, 0, 0, 0);
  if (date.getFullYear() !== year || date.getMonth() !== month - 1 || date.getDate() !== day) {
    throw new Error('Data inválida: ' + text);
  }
  return date;
}

function formatIsoDate(date) {
  if (!(date instanceof Date) || isNaN(date.getTime())) return '';
  return [
    String(date.getFullYear()).padStart(4, '0'),
    String(date.getMonth() + 1).padStart(2, '0'),
    String(date.getDate()).padStart(2, '0')
  ].join('-');
}

function parseBrMoney(value) {
  var original = String(value == null ? '' : value).trim();
  if (!original) throw new Error('Valor inválido: vazio');
  var normalized = original.replace(/R\$/gi, '').replace(/\s/g, '').replace(/\./g, '').replace(',', '.');
  if (!/^-?\d+(\.\d+)?$/.test(normalized)) throw new Error('Valor inválido: ' + original);
  var number = Number(normalized);
  if (!isFinite(number)) throw new Error('Valor inválido: ' + original);
  return number;
}

function normalizeKeySegment(value) {
  return String(value == null ? '' : value)
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toUpperCase()
    .trim()
    .replace(/[^A-Z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
}

function splitContract(fullContract) {
  var parts = String(fullContract == null ? '' : fullContract).trim().split(/\s+-\s+/);
  if (parts.length < 2) return { frequency: '', location: '' };
  return { frequency: parts[0].trim(), location: parts[1].trim() };
}

function buildContractKey(id, fullContract, startDate) {
  var cleanId = normalizeId(id);
  var contract = normalizeKeySegment(fullContract);
  var start = formatIsoDate(startDate);
  if (!cleanId || !contract || !start) throw new Error('Não foi possível criar a chave do contrato.');
  return cleanId + '|' + contract + '|' + start;
}
```

The date validator rejects `31/02/2026` by reconstructing the Date at noon and comparing year, month and day. `splitContract` splits only on hyphens surrounded by whitespace and returns empty frequency/location when fewer than two segments exist.

Create `src/03_Transformers.gs`:

```javascript
function buildMasterData(vencimentos, fichas, avaliacoes, importId) {
  var latestFicha = indexLatestById_(fichas, 'codigo', 'data inicio');
  var latestAvaliacao = indexLatestById_(avaliacoes, 'codigo', 'data da avaliacao');
  var studentsById = {};
  var contracts = [];
  var keys = {};
  var warnings = [];

  vencimentos.forEach(function (row) {
    var id = normalizeId(row.codigo);
    if (!id) throw new Error('ID vazio em vencimentos');
    if (!studentsById[id]) {
      var ficha = latestFicha[id] || null;
      var avaliacao = latestAvaliacao[id] || null;
      studentsById[id] = [
        id,
        String(row.cliente || '').trim(),
        ficha ? String(ficha.contato || '').trim() : '',
        String(row['status cliente'] || '').trim(),
        '',
        ficha ? parseBrDate(ficha['data inicio']) : '',
        avaliacao ? parseBrDate(avaliacao['data da avaliacao']) : '',
        importId
      ];
    }

    var fullContract = String(row.contrato || '').trim();
    var start = parseBrDate(row.inicio);
    var due = parseBrDate(row.vencimento);
    var split = splitContract(fullContract);
    if (!split.frequency && fullContract) warnings.push('Contrato sem frequência/polo: ID ' + id + ' - ' + fullContract);
    var key = buildContractKey(id, fullContract, start);
    if (keys[key]) throw new Error('Chave de contrato duplicada: ' + key);
    keys[key] = true;
    contracts.push([
      key, id, fullContract, split.frequency, parseBrMoney(row.valor), start, due,
      String(row['status contrato'] || '').trim(), split.location,
      String(row.modalidade || '').trim(), importId
    ]);
  });

  var students = Object.keys(studentsById).sort(function (a, b) { return Number(a) - Number(b); })
    .map(function (id) { return studentsById[id]; });
  var master = contracts.map(function (contract) {
    var student = studentsById[contract[1]];
    return [student[0], student[1], student[2], student[3], contract[3], contract[4], student[4], contract[5], contract[6], contract[8], student[5], student[6], contract[0]];
  });
  return { students: students, contracts: contracts, master: master, warnings: warnings };
}

function indexLatestById_(rows, idField, dateField) {
  return rows.reduce(function (index, row) {
    var id = normalizeId(row[idField]);
    if (!id) return index;
    var currentDate = parseBrDate(row[dateField]);
    if (!index[id] || currentDate.getTime() > parseBrDate(index[id][dateField]).getTime()) index[id] = row;
    return index;
  }, {});
}
```

```javascript
function getMasterSpreadsheet_() {
  return SpreadsheetApp.openById(CONFIG.spreadsheetId);
}

function ensureMasterSheets() {
  var ss = getMasterSpreadsheet_();
  Object.keys(CONFIG.sheets).forEach(function (key) {
    var name = CONFIG.sheets[key];
    var sheet = ss.getSheetByName(name) || ss.insertSheet(name);
    var headers = CONFIG.headers[key];
    if (sheet.getMaxColumns() < headers.length) {
      sheet.insertColumnsAfter(sheet.getMaxColumns(), headers.length - sheet.getMaxColumns());
    }
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold');
    sheet.setFrozenRows(1);
    var existing = sheet.getProtections(SpreadsheetApp.ProtectionType.RANGE).filter(function (protection) {
      return protection.getDescription() === 'TecnoFit - cabeçalho';
    });
    if (!existing.length) sheet.getRange(1, 1, 1, headers.length).protect().setDescription('TecnoFit - cabeçalho').setWarningOnly(true);
  });
  ss.getSheetByName(CONFIG.sheets.master).hideColumns(13);
}

function backupManagedSheets() {
  var ss = getMasterSpreadsheet_();
  return ['students', 'contracts', 'master'].reduce(function (backup, key) {
    var sheet = ss.getSheetByName(CONFIG.sheets[key]);
    backup[key] = sheet.getDataRange().getValues();
    return backup;
  }, {});
}

function restoreManagedSheets(backup) {
  var ss = getMasterSpreadsheet_();
  ['students', 'contracts', 'master'].forEach(function (key) {
    var sheet = ss.getSheetByName(CONFIG.sheets[key]);
    sheet.clearContents();
    var values = backup[key] || [CONFIG.headers[key]];
    if (values.length && values[0].length) sheet.getRange(1, 1, values.length, values[0].length).setValues(values);
  });
  SpreadsheetApp.flush();
}

function validateRowWidths_(rows, width, label) {
  rows.forEach(function (row, index) {
    if (row.length !== width) throw new Error(label + ': largura inválida na linha ' + (index + 2));
  });
}

function writeManagedSheet_(sheet, headers, rows) {
  validateRowWidths_(rows, headers.length, sheet.getName());
  sheet.clearContents();
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold');
  if (rows.length) sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  sheet.setFrozenRows(1);
  var oldFilter = sheet.getFilter();
  if (oldFilter) oldFilter.remove();
  sheet.getRange(1, 1, Math.max(rows.length + 1, 2), headers.length).createFilter();
}

function replaceManagedSheets(data) {
  validateRowWidths_(data.students, CONFIG.headers.students.length, CONFIG.sheets.students);
  validateRowWidths_(data.contracts, CONFIG.headers.contracts.length, CONFIG.sheets.contracts);
  validateRowWidths_(data.master, CONFIG.headers.master.length, CONFIG.sheets.master);

  var ss = getMasterSpreadsheet_();
  var students = ss.getSheetByName(CONFIG.sheets.students);
  var contracts = ss.getSheetByName(CONFIG.sheets.contracts);
  var master = ss.getSheetByName(CONFIG.sheets.master);
  writeManagedSheet_(students, CONFIG.headers.students, data.students);
  writeManagedSheet_(contracts, CONFIG.headers.contracts, data.contracts);
  writeManagedSheet_(master, CONFIG.headers.master, data.master);

  students.getRange('F:G').setNumberFormat('dd/MM/yyyy');
  contracts.getRange('E:E').setNumberFormat('R$ #,##0.00');
  contracts.getRange('F:G').setNumberFormat('dd/MM/yyyy');
  master.getRange('F:F').setNumberFormat('R$ #,##0.00');
  master.getRange('H:I').setNumberFormat('dd/MM/yyyy');
  master.getRange('K:L').setNumberFormat('dd/MM/yyyy');
  master.hideColumns(13);
  SpreadsheetApp.flush();
}
```

`backupManagedSheets` returns values for the three managed data tabs, including headers. Row widths are validated before any clear operation. `IMPORTACOES` remains untouched.

```javascript
var SOURCE_HEADERS = Object.freeze({
  vencimentos: ['codigo', 'cliente', 'status cliente', 'contrato', 'valor', 'inicio', 'vencimento', 'status contrato', 'modalidade'],
  fichas: ['codigo', 'data inicio', 'contato'],
  avaliacao_fisica: ['codigo', 'data da avaliacao']
});

function parseImportFilename(name) {
  var match = CONFIG.filenamePattern.exec(String(name || ''));
  if (!match) throw new Error('Nome inválido: ' + name);
  return { type: match[1].toLowerCase(), referenceDate: match[2], revision: match[3] };
}

function groupInputBatch_(files) {
  if (!files.length) throw new Error('01_ENTRADA está vazia.');
  var groups = {};
  files.forEach(function (entry) {
    var parsed = parseImportFilename(entry.name);
    var groupKey = parsed.referenceDate + '|r' + parsed.revision;
    groups[groupKey] = groups[groupKey] || {
      referenceDate: parsed.referenceDate,
      revision: parsed.revision,
      filesByType: {}
    };
    if (groups[groupKey].filesByType[parsed.type]) throw new Error('Arquivo repetido no lote: ' + parsed.type);
    groups[groupKey].filesByType[parsed.type] = entry;
  });
  var keys = Object.keys(groups);
  if (keys.length !== 1) throw new Error('Há mais de um lote em 01_ENTRADA.');
  var batch = groups[keys[0]];
  var missing = CONFIG.requiredTypes.filter(function (type) { return !batch.filesByType[type]; });
  if (missing.length) throw new Error('Lote incompleto. Faltando: ' + missing.join(', '));
  return batch;
}

function findSingleInputBatch() {
  var iterator = DriveApp.getFolderById(CONFIG.inputFolderId).getFiles();
  var files = [];
  while (iterator.hasNext()) {
    var file = iterator.next();
    files.push({ id: file.getId(), name: file.getName(), file: file });
  }
  return groupInputBatch_(files);
}

function readBatchTables(batch) {
  var result = { counts: {} };
  CONFIG.requiredTypes.forEach(function (type) {
    var entry = batch.filesByType[type];
    var html = entry.file.getBlob().getDataAsString('UTF-8');
    var objects = tableToObjects(parseHtmlTable(html), SOURCE_HEADERS[type]);
    result[type] = objects;
    result.counts[type] = { read: objects.length, valid: objects.length, rejected: 0 };
  });
  return result;
}

function getOrCreateChildFolder_(parent, name) {
  var folders = parent.getFoldersByName(name);
  return folders.hasNext() ? folders.next() : parent.createFolder(name);
}

function moveBatch(batch, destinationRootId) {
  var root = DriveApp.getFolderById(destinationRootId);
  var yearFolder = getOrCreateChildFolder_(root, batch.referenceDate.slice(0, 4));
  var dateFolder = getOrCreateChildFolder_(yearFolder, batch.referenceDate);
  CONFIG.requiredTypes.forEach(function (type) {
    batch.filesByType[type].file.moveTo(dateFolder);
  });
  return dateFolder.getId();
}
```

```javascript
function startImportLog(batch, executionId) {
  var sheet = getMasterSpreadsheet_().getSheetByName(CONFIG.sheets.imports);
  var start = new Date();
  var firstRow = sheet.getLastRow() + 1;
  var values = CONFIG.requiredTypes.map(function (type) {
    var entry = batch.filesByType[type];
    return [executionId, start, '', type, entry.name, entry.id, batch.referenceDate, 'r' + batch.revision, '', '', '', 'PROCESSANDO', 'Importação iniciada'];
  });
  sheet.getRange(firstRow, 1, values.length, CONFIG.headers.imports.length).setValues(values);
  SpreadsheetApp.flush();
  return CONFIG.requiredTypes.map(function (type, index) { return { type: type, row: firstRow + index }; });
}

function finishImportLog(logRows, result) {
  var sheet = getMasterSpreadsheet_().getSheetByName(CONFIG.sheets.imports);
  var end = new Date();
  logRows.forEach(function (logRow) {
    var counts = (result.counts && result.counts[logRow.type]) || { read: '', valid: '', rejected: '' };
    sheet.getRange(logRow.row, 3).setValue(end);
    sheet.getRange(logRow.row, 9, 1, 5).setValues([[
      counts.read,
      counts.valid,
      counts.rejected,
      result.status,
      String(result.message || '').slice(0, 500)
    ]]);
  });
  SpreadsheetApp.flush();
}
```

```javascript
function runImport() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) throw new Error('Já existe uma importação em andamento.');
  var batch = null;
  var logRows = null;
  var backup = null;
  var executionId = Utilities.getUuid();
  try {
    batch = findSingleInputBatch();
    logRows = startImportLog(batch, executionId);
    var tables = readBatchTables(batch);
    var data = buildMasterData(tables.vencimentos, tables.fichas, tables.avaliacao_fisica, executionId);
    backup = backupManagedSheets();
    replaceManagedSheets(data);
    moveBatch(batch, CONFIG.processedFolderId);
    finishImportLog(logRows, {
      status: 'SUCESSO',
      message: data.warnings.length ? data.warnings.join(' | ') : 'Base atualizada',
      counts: tables.counts
    });
    return { executionId: executionId, students: data.students.length, contracts: data.contracts.length, warnings: data.warnings };
  } catch (error) {
    if (backup) restoreManagedSheets(backup);
    if (batch) {
      try { moveBatch(batch, CONFIG.rejectedFolderId); } catch (moveError) { console.error(moveError); }
    }
    if (logRows) finishImportLog(logRows, { status: 'ERRO', message: String(error.message || error), counts: {} });
    throw error;
  } finally {
    lock.releaseLock();
  }
}
```
